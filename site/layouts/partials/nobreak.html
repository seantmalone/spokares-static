{{- /* Keeps form and course numbers such as "ICS-213RR" and "IS-700" (with any brackets around them) on one line (no break at the hyphen) in rendered HTML: each one in a text node is wrapped in . Usage: partial "nobreak.html" $html. Used for Markdown blocks and the Resources list. */ -}} {{- $out := string . -}} {{- range uniq (findRE `>[^<]*\b(?:ICS|IS)-\d[^<]*<` $out) -}} {{- $out = replace $out . (replaceRE `(\(?)\b(ICS|IS)-(\d+[A-Z]*)(\)?)` `${1}${2}-${3}${4}` .) -}} {{- end -}} {{- return ($out | safeHTML) -}}
