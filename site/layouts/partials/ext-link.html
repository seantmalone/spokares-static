(urls.Parse $url).Hostname) -}} {{- $class := trim (printf "%s ext" (or .class "")) " " -}} {{- return (printf `%s (opens %s)` $class (htmlEscape $url) (htmlEscape $text) (htmlEscape $opens) | safeHTML) -}}
