{{- /* A link to another site, marked the way the theme marked them: class "ext" (the small arrow) and a visually hidden "(opens …)" for screen readers. {{ partial "ext-link.html" (dict "url" "https://www.weather.gov/otx/" "text" "NWS Spokane" "opens" "the National Weather Service" "class" "doc-title") }} opens is optional; without it the label names the host (www. dropped). class adds classes beside "ext". file names the data file the address came from, for the error message. The arrow is drawn after the last word (span.ext__tail), which never wraps away from it. */ -}} {{- $file := .file | default "the page" -}} {{- $url := string (or .url "") -}} {{- $text := string (or .text "") -}} {{- if not $text }}{{ errorf "%s: a link to %q has no title" $file $url }}{{ end -}} {{- if not (findRE `^https?://[^\s"'<>]+$` $url) }}{{ errorf "%s: %q is not a web address starting with https://" $file $url }}{{ end -}} {{- with findRE `%(?:[^0-9A-Fa-f]|[0-9A-Fa-f][^0-9A-Fa-f]|[0-9A-Fa-f]?$)` $url 1 -}} {{- errorf "%s: the address %q has a %% sign that is not followed by two hex digits; copy the address again from the page" $file $url -}} {{- end -}} {{- $opens := or .opens (strings.TrimPrefix "www." (urls.Parse $url).Hostname) -}} {{- $class := trim (printf "%s ext" (or .class "")) " " -}} {{- /* The last word and the arrow stay together on one line (.ext__tail). A button (class btn) is one flex line already and keeps the plain form. */ -}} {{- $inner := htmlEscape $text -}} {{- if not (in (split $class " ") "btn") -}} {{- $words := split $text " " -}} {{- $last := index $words (sub (len $words) 1) -}} {{- $inner = printf `%s%s` (htmlEscape (strings.TrimSuffix $last $text)) (htmlEscape $last) -}} {{- end -}} {{- return (printf `%s (opens %s)` $class (htmlEscape $url) $inner (htmlEscape $opens) | safeHTML) -}}
